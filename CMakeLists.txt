cmake_minimum_required(VERSION 4.0)
project(JevSimulation)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "Build raylib examples" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build static libraries" FORCE)
FetchContent_Declare(raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG 6.0
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(raylib)

set(FLECS_STATIC ON CACHE BOOL "Build static Flecs" FORCE)
set(FLECS_SHARED OFF CACHE BOOL "Disable shared Flecs" FORCE)
set(FLECS_TESTS OFF CACHE BOOL "Disable dependency tests" FORCE)
FetchContent_Declare(flecs
    GIT_REPOSITORY https://github.com/SanderMertens/flecs.git
    GIT_TAG v4.1.6
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(flecs)

add_library(JevWorld STATIC Grid.cpp WorldDocument.cpp SceneWorld.cpp Guid.cpp Object.cpp Entity.cpp GridNavigation.cpp WorldCamera.cpp TilesetLibrary.cpp TilesetEditor.cpp ObjectInspector.cpp CatalogEditor.cpp MapStorage.cpp AtomicFile.cpp)
target_include_directories(JevWorld PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_sources(JevWorld PRIVATE Resource.cpp Position.cpp Range.cpp Decision.cpp EntityPresentation.cpp CreatureCatalog.cpp CreatureEditor.cpp)
target_sources(JevWorld PRIVATE Harvestable.cpp HarvestEditor.cpp)
target_sources(JevWorld PRIVATE PlayerController.cpp)
target_link_libraries(JevWorld PUBLIC raylib flecs_static)
add_executable(JevSimulation main.cpp NodeProcess.cpp DecisionBridge.cpp AIDemo.cpp MemoryUsage.cpp)
target_link_libraries(JevSimulation PRIVATE JevWorld)
if(WIN32)
    target_link_libraries(JevSimulation PRIVATE psapi)
endif()

find_program(JEV_NODE_EXECUTABLE NAMES node REQUIRED)
configure_file(NodeConfig.h.in generated/NodeConfig.h @ONLY)
target_include_directories(JevSimulation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MINGW)
    # Also run from a terminal without CLion's injected compiler DLL paths.
    target_link_options(JevSimulation PRIVATE -static)
endif()
